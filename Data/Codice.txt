import numpy as np
from scipy.signal import hilbert

class NeuroCoreKernel:
    """
    Minimal NeuroCore kernel.

    This implementation does NOT assume that the signal follows
    the full Kuramoto dynamical model.

    The Kuramoto order parameter r(t) is used only as an empirical
    phase-coherence observable across channels.
    """

    def __init__(self, sampling_rate):
        self.fs = sampling_rate

        # Empirical omega regimes from current validation.
        # These are observational regimes, not clinical diagnoses.
        self.REGIMES = {
            "HIGH_ENTROPY": (0.00, 0.35),
            "TRANSITION":  (0.35, 0.65),
            "RIGID_SYNC":  (0.65, 1.00),
        }

    def compute_phi_operator(self, signal):
        """
        Structural operator Phi.

        Phi(t) = mean log(1 + |gradient|)
        Normalized by length to reduce dependence on window size.
        """
        signal = np.asarray(signal, dtype=float)

        if signal.ndim > 1:
            signal = signal.reshape(-1)

        gradient = np.abs(np.diff(signal))
        phi = np.mean(np.log1p(gradient))

        return float(phi)

    def compute_delta_operator(self, phi_series):
        """
        Transition-sensitivity operator.

        Delta(t) = |Phi(t) - Phi(t-1)|
        """
        phi_series = np.asarray(phi_series, dtype=float)
        delta = np.abs(np.diff(phi_series, prepend=phi_series[0]))
        return delta

    def compute_entropy(self, signal, bins=64):
        """
        Normalized Shannon entropy in [0, 1].
        """
        signal = np.asarray(signal, dtype=float).reshape(-1)

        hist, _ = np.histogram(signal, bins=bins)
        p = hist / (np.sum(hist) + 1e-12)
        p = p[p > 0]

        h = -np.sum(p * np.log(p))
        h_norm = h / np.log(bins)

        return float(np.clip(h_norm, 0.0, 1.0))

    def get_kuramoto_order(self, multichannel_data):
        """
        Empirical Kuramoto order parameter.

        Input shape:
        channels x samples
        """
        X = np.asarray(multichannel_data, dtype=float)

        if X.ndim != 2:
            raise ValueError("multichannel_data must have shape: channels x samples")

        phases = np.angle(hilbert(X, axis=1))
        r_t = np.abs(np.mean(np.exp(1j * phases), axis=0))

        return r_t

    def compute_omega(self, r_value, h_entropy):
        """
        Composite NeuroCore observable.

        omega = r * (1 - h)
        """
        r_value = float(np.clip(r_value, 0.0, 1.0))
        h_entropy = float(np.clip(h_entropy, 0.0, 1.0))

        omega = r_value * (1.0 - h_entropy)

        return float(np.clip(omega, 0.0, 1.0))

    def classify_state(self, omega):
        """
        Classify observational regime from omega.

        This is NOT a clinical diagnosis.
        """
        omega = float(np.clip(omega, 0.0, 1.0))

        if omega < self.REGIMES["HIGH_ENTROPY"][1]:
            return "HIGH_ENTROPY_REGIME"
        elif omega < self.REGIMES["TRANSITION"][1]:
            return "TRANSITION_REGIME"
        else:
            return "RIGID_SYNCHRONIZATION_REGIME"

    def analyze_window(self, multichannel_window):
        """
        Full window-level analysis.

        Returns:
        entropy h
        mean phase coherence r
        omega
        phi
        regime label
        """
        X = np.asarray(multichannel_window, dtype=float)

        h = self.compute_entropy(X)
        r_t = self.get_kuramoto_order(X)
        r_mean = float(np.mean(r_t))

        omega = self.compute_omega(r_mean, h)
        phi = self.compute_phi_operator(X)
        regime = self.classify_state(omega)

        return {
            "entropy": h,
            "r": r_mean,
            "omega": omega,
            "phi": phi,
            "regime": regime
        }
