// neurocore_kernel.c
// Minimal NeuroCore Kernel in C
// gcc neurocore_kernel.c -o neurocore -lm

#include <stdio.h>
#include <stdlib.h>
#include <math.h>

typedef enum {
    HIGH_ENTROPY_REGIME,
    TRANSITION_REGIME,
    RIGID_SYNCHRONIZATION_REGIME
} NeuroCoreRegime;

double clamp(double x, double min, double max) {
    if (x < min) return min;
    if (x > max) return max;
    return x;
}

double compute_phi_operator(const double *signal, int n) {
    if (n < 2) return 0.0;

    double sum = 0.0;

    for (int i = 1; i < n; i++) {
        double grad = fabs(signal[i] - signal[i - 1]);
        sum += log1p(grad);
    }

    return sum / (double)(n - 1);
}

void compute_delta_operator(const double *phi_series, double *delta, int n) {
    if (n <= 0) return;

    delta[0] = 0.0;

    for (int i = 1; i < n; i++) {
        delta[i] = fabs(phi_series[i] - phi_series[i - 1]);
    }
}

double compute_entropy(const double *signal, int n, int bins) {
    if (n <= 0 || bins <= 1) return 0.0;

    double min_val = signal[0];
    double max_val = signal[0];

    for (int i = 1; i < n; i++) {
        if (signal[i] < min_val) min_val = signal[i];
        if (signal[i] > max_val) max_val = signal[i];
    }

    double range = max_val - min_val;
    if (range == 0.0) return 0.0;

    int *hist = calloc(bins, sizeof(int));
    if (!hist) return 0.0;

    for (int i = 0; i < n; i++) {
        int idx = (int)(((signal[i] - min_val) / range) * bins);
        if (idx >= bins) idx = bins - 1;
        if (idx < 0) idx = 0;
        hist[idx]++;
    }

    double h = 0.0;

    for (int k = 0; k < bins; k++) {
        if (hist[k] > 0) {
            double p = (double)hist[k] / (double)n;
            h -= p * log(p);
        }
    }

    free(hist);

    return clamp(h / log((double)bins), 0.0, 1.0);
}

double compute_omega(double r_value, double entropy) {
    r_value = clamp(r_value, 0.0, 1.0);
    entropy = clamp(entropy, 0.0, 1.0);

    return clamp(r_value * (1.0 - entropy), 0.0, 1.0);
}

NeuroCoreRegime classify_state(double omega) {
    omega = clamp(omega, 0.0, 1.0);

    if (omega < 0.35) {
        return HIGH_ENTROPY_REGIME;
    } else if (omega < 0.65) {
        return TRANSITION_REGIME;
    } else {
        return RIGID_SYNCHRONIZATION_REGIME;
    }
}

const char* regime_to_string(NeuroCoreRegime regime) {
    switch (regime) {
        case HIGH_ENTROPY_REGIME:
            return "HIGH_ENTROPY_REGIME";
        case TRANSITION_REGIME:
            return "TRANSITION_REGIME";
        case RIGID_SYNCHRONIZATION_REGIME:
            return "RIGID_SYNCHRONIZATION_REGIME";
        default:
            return "UNKNOWN";
    }
}

int main() {
    double signal[] = {
        0.1, 0.2, 0.15, 0.4, 0.9, 1.2, 0.8, 0.3, 0.2, 0.1
    };

    int n = sizeof(signal) / sizeof(signal[0]);

    double phi = compute_phi_operator(signal, n);
    double entropy = compute_entropy(signal, n, 64);

    // Placeholder: in the minimal C kernel, r must come from an external
    // phase-coherence module or FFT/Hilbert implementation.
    double r_observed = 0.80;

    double omega = compute_omega(r_observed, entropy);
    NeuroCoreRegime regime = classify_state(omega);

    printf("Phi: %.6f\n", phi);
    printf("Entropy: %.6f\n", entropy);
    printf("r: %.6f\n", r_observed);
    printf("Omega: %.6f\n", omega);
    printf("Regime: %s\n", regime_to_string(regime));

    return 0;
}
