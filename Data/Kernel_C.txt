#include "main.h"
#include <math.h>
#include <stdio.h>
#include <string.h>

/* --- CONFIGURAZIONE KERNEL --- */
#define WINDOW_SIZE 64       // Dimensione finestra mobile (W)
#define THRESHOLD_KC 1.441   // Soglia Critica Universale
#define ATTRACTOR_PHI 0.55   // Punto di equilibrio dinamico

/* --- VARIABILI GLOBALI --- */
float signal_buffer[WINDOW_SIZE];
int buffer_index = 0;
UART_HandleTypeDef huart2; // Assicurati che UART sia configurata nel tuo progetto

/* --- 1. OPERATORE DI FLUSSO ENTROPICO (Phi) --- */
float compute_phi(float* buffer, int n) {
    float phi_sum = 0.0f;
    for (int i = 1; i < n; i++) {
        float grad = fabsf(buffer[i] - buffer[i-1]);
        phi_sum += log1pf(grad); // Linearizzazione della densità di info
    }
    return phi_sum / (float)(n - 1);
}

/* --- 2. OPERATORE DI STABILITÀ (K - Resilience Index) --- */
float compute_k_resilience(float* buffer, int n) {
    float sum = 0, mean, std_dev = 0;
    for (int i = 0; i < n; i++) sum += buffer[i];
    mean = sum / n;

    for (int i = 0; i < n; i++) 
        std_dev += powf(buffer[i] - mean, 2);
    
    std_dev = sqrtf(std_dev / n);
    
    // Ritorna il coefficiente di variazione (Proxy della sincronizzazione)
    return std_dev / (fabsf(mean) + 0.0001f);
}

/* --- 3. TRANSITION SCORE UNIFICATO (TS) --- */
float calculate_ts(float phi, float k) {
    // Il TS esplode quando Phi (entropia) e K (resilienza) crollano
    float ts = 1.0f / (phi * k + 0.1f); 
    return ts;
}

/* --- CICLO DI ELABORAZIONE --- */
void process_neurocore_step(float new_sample) {
    // Inserimento nuovo dato nel buffer circolare
    signal_buffer[buffer_index] = new_sample;
    buffer_index = (buffer_index + 1) % WINDOW_SIZE;

    // Calcolo degli operatori
    float current_phi = compute_phi(signal_buffer, WINDOW_SIZE);
    float current_k = compute_k_resilience(signal_buffer, WINDOW_SIZE);
    float ts = calculate_ts(current_phi, current_k);

    // Classificazione Regime
    char* regime;
    if (ts < 0.8f) regime = "STABLE";
    else if (ts < THRESHOLD_KC) regime = "TRANSITION";
    else regime = "CRITICAL_COLLAPSE";

    // Output verso il PC (Serial Plotter)
    char msg[128];
    sprintf(msg, "Phi:%.3f, K:%.3f, TS:%.3f, State:%s\r\n", current_phi, current_k, ts, regime);
    HAL_UART_Transmit(&huart2, (uint8_t*)msg, strlen(msg), 10);
}

/* --- MAIN LOOP --- */
int main(void) {
    HAL_Init();
    SystemClock_Config(); // Configura il clock al massimo (es. 72MHz-168MHz)
    MX_USART2_UART_Init();
    MX_ADC1_Init();

    while (1) {
        // Esempio: Leggi da ADC o ricevi da Seriale
        uint32_t adc_val = HAL_ADC_GetValue(&hadc1);
        float voltage = (adc_val * 3.3f) / 4095.0f;

        process_neurocore_step(voltage);

        HAL_Delay(10); // Campionamento a 100Hz
    }
}
